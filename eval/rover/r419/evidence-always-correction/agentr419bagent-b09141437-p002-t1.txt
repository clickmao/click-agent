#!/usr/bin/env python3
"""单文件 JSON 规范化器（严格子集，纯标准库）。

用法:  python3 solution.py < input.txt > output.txt
约定:  读 STDIN 整体作为一个 JSON 值；输出单行（无任何空白）规范化文本；
       任何非法输入（语法错误/多余尾随内容/非法转义/非法数字/空输入）只输出 ERR。
       不打印任何额外文字、提示或调试信息。

支持的范围（严格按题面）:
  null / true / false
  十进制整数（可带一个前导 '-'；不允许前导零；-0 输出为 0；不接受 . e +）
  字符串（六种转义: \\" \\\\ \\/ \\n \\t \\uXXXX；\\\\u 解码码点必须 >= 0x20）
  数组、对象（键为字符串；重复键后者覆盖前者；输出按键的 Unicode 码点升序）
输出转义: 仅把 引号 / 反斜杠 / 换行 / 制表符 重新转义，其余字符原样。
"""

import sys

_WS = " \t\n\r"
_HEX = "0123456789abcdefABCDEF"

# 允许较深嵌套；纯 Python 递归在 3.11+ 不额外吃 C 栈。
try:
    sys.setrecursionlimit(20000)
except Exception:  # pragma: no cover
    pass


class _Err(Exception):
    """任何语法错误统一用它打断解析，由 main 兜底成 ERR。"""


class _Parser:
    __slots__ = ("s", "i", "n")

    def __init__(self, s):
        self.s = s
        self.i = 0
        self.n = len(s)

    # ---------- 基础工具 ----------
    def _fail(self):
        raise _Err()

    def _ws(self):
        s, n, i = self.s, self.n, self.i
        while i < n and s[i] in _WS:
            i += 1
        self.i = i

    def _expect(self, word):
        if self.s[self.i:self.i + len(word)] != word:
            self._fail()
        self.i += len(word)

    def _hex4(self):
        if self.i + 4 > self.n:
            self._fail()
        h = self.s[self.i:self.i + 4]
        for ch in h:
            if ch not in _HEX:
                self._fail()
        self.i += 4
        return int(h, 16)

    # ---------- 顶层 ----------
    def parse(self):
        self._ws()
        if self.i >= self.n:
            self._fail()            # 空输入
        v = self._value()
        self._ws()
        if self.i != self.n:        # 结尾多余内容
            self._fail()
        return v

    # ---------- 值 ----------
    def _value(self):
        if self.i >= self.n:
            self._fail()
        c = self.s[self.i]
        if c == '"':
            return self._string()
        if c == '[':
            return self._array()
        if c == '{':
            return self._object()
        if c == 'n':
            self._expect('null')
            return None
        if c == 't':
            self._expect('true')
            return True
        if c == 'f':
            self._expect('false')
            return False
        if c == '-' or ('0' <= c <= '9'):
            return self._number()
        self._fail()

    # ---------- 数字（只允许整数） ----------
    def _number(self):
        s, n = self.s, self.n
        start = self.i
        if s[self.i] == '-':
            self.i += 1
            if self.i >= n or not ('0' <= s[self.i] <= '9'):
                self._fail()
        if s[self.i] == '0':
            self.i += 1                     # 前导零只允许单独一个 0
        else:
            while self.i < n and '0' <= s[self.i] <= '9':
                self.i += 1
        # 后面的字符由调用方校验（必须是分隔符/结尾），此处只管取数
        return int(s[start:self.i])

    # ---------- 字符串 ----------
    def _string(self):
        s, n = self.s, self.n
        self.i += 1                          # 跳过起始引号
        codes = []
        while True:
            if self.i >= n:
                self._fail()                 # 未闭合
            c = s[self.i]
            if c == '"':
                self.i += 1
                return _join_codes(codes)
            if c == '\\':
                self.i += 1
                if self.i >= n:
                    self._fail()
                e = s[self.i]
                if e == '"':
                    codes.append(0x22); self.i += 1
                elif e == '\\':
                    codes.append(0x5C); self.i += 1
                elif e == '/':
                    codes.append(0x2F); self.i += 1
                elif e == 'n':
                    codes.append(0x0A); self.i += 1
                elif e == 't':
                    codes.append(0x09); self.i += 1
                elif e == 'u':
                    self.i += 1
                    cp = self._hex4()
                    if cp < 0x20:        # 解码出的码点必须 >= 0x20
                        self._fail()
                    codes.append(cp)
                else:
                    self._fail()         # 非六种标准转义之一
            else:
                if c < ' ':              # 字符串内禁止裸控制字符
                    self._fail()
                codes.append(ord(c))
                self.i += 1

    # ---------- 数组 ----------
    def _array(self):
        self.i += 1                          # '['
        self._ws()
        arr = []
        if self.i < self.n and self.s[self.i] == ']':
            self.i += 1
            return arr
        while True:
            arr.append(self._value())
            self._ws()
            if self.i >= self.n:
                self._fail()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
                self._ws()
            elif c == ']':
                self.i += 1
                return arr
            else:
                self._fail()

    # ---------- 对象 ----------
    def _object(self):
        self.i += 1                          # '{'
        d = {}
        self._ws()
        if self.i < self.n and self.s[self.i] == '}':
            self.i += 1
            return d
        while True:
            self._ws()
            if self.i >= self.n or self.s[self.i] != '"':
                self._fail()                 # 键必须是字符串
            k = self._string()
            self._ws()
            if self.i >= self.n or self.s[self.i] != ':':
                self._fail()
            self.i += 1
            self._ws()
            d[k] = self._value()             # 重复键：后者覆盖前者
            self._ws()
            if self.i >= self.n:
                self._fail()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
            elif c == '}':
                self.i += 1
                return d
            else:
                self._fail()


def _join_codes(codes):
    """把码点序列拼成字符串；相邻的 UTF-16 代理对合并为一个码点。"""
    out = []
    k = 0
    m = len(codes)
    while k < m:
        c = codes[k]
        if 0xD800 <= c <= 0xDBFF and k + 1 < m and 0xDC00 <= codes[k + 1] <= 0xDFFF:
            out.append(chr(0x10000 + ((c - 0xD800) << 10) + (codes[k + 1] - 0xDC00)))
            k += 2
        else:
            out.append(chr(c))
            k += 1
    return "".join(out)


def _esc(text):
    """输出转义：只重新转义 引号 / 反斜杠 / 换行 / 制表符。"""
    out = []
    for ch in text:
        if ch == '"':
            out.append('\\"')
        elif ch == '\\':
            out.append('\\\\')
        elif ch == '\n':
            out.append('\\n')
        elif ch == '\t':
            out.append('\\t')
        else:
            out.append(ch)
    return "".join(out)


def _dumps(v):
    """规范化序列化：无任何空白；对象键按 Unicode 码点升序。"""
    if v is None:
        return 'null'
    if v is True:
        return 'true'
    if v is False:
        return 'false'
    if isinstance(v, int):
        return str(v)
    if isinstance(v, str):
        return '"' + _esc(v) + '"'
    if isinstance(v, list):
        return '[' + ','.join(_dumps(x) for x in v) + ']'
    items = sorted(v.items(), key=lambda kv: kv[0])
    return '{' + ','.join('"' + _esc(k) + '":' + _dumps(x) for k, x in items) + '}'


def main():
    raw = sys.stdin.buffer.read()
    try:
        text = raw.decode('utf-8')
    except UnicodeDecodeError:
        sys.stdout.write('ERR')
        return

    try:
        value = _Parser(text).parse()
        result = _dumps(value)
    except (_Err, RecursionError):
        sys.stdout.write('ERR')
        return

    # 可能存在孤立代理码点：用 surrogatepass 保证不因编码异常而崩溃
    sys.stdout.buffer.write(result.encode('utf-8', 'surrogatepass'))
    sys.stdout.buffer.flush()


if __name__ == '__main__':
    main()

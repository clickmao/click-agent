DIGITS = "0123456789"
HEX = "0123456789abcdefABCDEF"


def skip_ws(s, i, n):
    while i < n and s[i] in WS:
        i += 1
    return i


def esc_out(ch):
    """Re-escape one character for the compact output form."""
    if ch == '"':
        return '\\"'
    if ch == '\\':
        return '\\\\'
    if ch == '\n':
        return '\\n'
    if ch == '\t':
        return '\\t'
    return ch

def read_number(s, i, n):
    """s[i] is '-' or a digit.  Return (normalized, next_index) or (None, i)."""
    start = i
    if s[i] == '-':
        i += 1
        if i >= n or s[i] not in DIGITS:
            return None, start
    if s[i] == '0':
        i += 1                     # no leading zeros ("0" / "-0" only)
    else:
        while i < n and s[i] in DIGITS:
            i += 1
    return str(int(s[start:i])), i  # int() normalizes "-0" to "0"

def process(raw):
    """raw = stdin bytes; return the bytes to write (without the newline)."""
    try:
        s = raw.decode('utf-8')
    except UnicodeDecodeError:
        return b'ERR'
    try:
        out = normalize(s)
    except Exception:
        out = None
    if out is None:
        return b'ERR'
    try:
        return out.encode('utf-8')
    except UnicodeEncodeError:
        return b'ERR'

import sys


class ParseError(Exception):
    pass


WS = ' \t\n\r'


class Parser:
    __slots__ = ('s', 'i', 'n')

    def __init__(self, s):
        self.s = s
        self.i = 0
        self.n = len(s)

    def skip_ws(self):
        s, n, i = self.s, self.n, self.i
        while i < n and s[i] in WS:
            i += 1
        self.i = i

    def parse(self):
        self.skip_ws()
        v = self.value()
        self.skip_ws()
        if self.i != self.n:
            raise ParseError()          # 结尾有多余内容
        return v

    def value(self):
        if self.i >= self.n:
            raise ParseError()
        c = self.s[self.i]
        if c == '{':
            return self.obj()
        if c == '[':
            return self.arr()
        if c == '"':
            return self.string()
        if c == 't':
            self.lit('true')
            return True
        if c == 'f':
            self.lit('false')
            return False
        if c == 'n':
            self.lit('null')
            return None
        if c == '-' or '0' <= c <= '9':
            return self.number()
        raise ParseError()

    def lit(self, word):
        if self.s[self.i:self.i + len(word)] != word:
            raise ParseError()
        self.i += len(word)

    def number(self):
        """十进制整数：可选负号，不允许前导零，-0 归一为 0。"""
        start = self.i
        if self.s[self.i] == '-':
            self.i += 1
        if self.i >= self.n or not ('0' <= self.s[self.i] <= '9'):
            raise ParseError()
        if self.s[self.i] == '0':
            self.i += 1
        else:
            while self.i < self.n and '0' <= self.s[self.i] <= '9':
                self.i += 1
        return int(self.s[start:self.i])

    def hex4(self):
        if self.i + 4 > self.n:
            raise ParseError()
        h = self.s[self.i:self.i + 4]
        for ch in h:
            if not ('0' <= ch <= '9' or 'a' <= ch <= 'f' or 'A' <= ch <= 'F'):
                raise ParseError()
        self.i += 4
        return int(h, 16)

    def string(self):
        """仅允许 \\\" \\\\ \\/ \\n \\t \\uXXXX 六种转义。"""
        self.i += 1                      # 跳过开引号
        out = []
        while True:
            if self.i >= self.n:
                raise ParseError()
            c = self.s[self.i]
            if c == '"':
                self.i += 1
                return ''.join(out)
            if c == '\\':
                self.i += 1
                if self.i >= self.n:
                    raise ParseError()
                e = self.s[self.i]
                if e == '"':
                    out.append('"')
                elif e == '\\':
                    out.append('\\')
                elif e == '/':
                    out.append('/')
                elif e == 'n':
                    out.append('\n')
                elif e == 't':
                    out.append('\t')
                elif e == 'u':
                    self.i += 1
                    cp = self.hex4()
                    # 码点必须 >= 0x20；代理区无法产出合法 Unicode 文本
                    if cp < 0x20 or 0xD800 <= cp <= 0xDFFF:
                        raise ParseError()
                    out.append(chr(cp))
                    continue
                else:
                    raise ParseError()   # 非法转义（如 \b \f \r \x 等）
                self.i += 1
            else:
                if ord(c) < 0x20:
                    raise ParseError()   # 字符串内裸控制字符
                out.append(c)
                self.i += 1

    def arr(self):
        self.i += 1
        a = []
        self.skip_ws()
        if self.i < self.n and self.s[self.i] == ']':
            self.i += 1
            return a
        while True:
            self.skip_ws()
            a.append(self.value())
            self.skip_ws()
            if self.i >= self.n:
                raise ParseError()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
                continue
            if c == ']':
                self.i += 1
                return a
            raise ParseError()

    def obj(self):
        self.i += 1
        d = {}
        self.skip_ws()
        if self.i < self.n and self.s[self.i] == '}':
            self.i += 1
            return d
        while True:
            self.skip_ws()
            if self.i >= self.n or self.s[self.i] != '"':
                raise ParseError()
            k = self.string()
            self.skip_ws()
            if self.i >= self.n or self.s[self.i] != ':':
                raise ParseError()
            self.i += 1
            self.skip_ws()
            v = self.value()
            d[k] = v                     # 重复键：后者覆盖前者
            self.skip_ws()
            if self.i >= self.n:
                raise ParseError()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
                continue
            if c == '}':
                self.i += 1
                return d
            raise ParseError()


def escape(s):
    """只重新转义 引号 / 反斜杠 / 换行 / 制表符，其余字符原样。"""
    out = []
    for ch in s:
        if ch == '"':
            out.append('\\"')
        elif ch == '\\':
            out.append('\\\\')
        elif ch == '\n':
            out.append('\\n')
        elif ch == '\t':
            out.append('\\t')
        else:
            out.append(ch)
    return ''.join(out)


def dump(v, out):
    if v is None:
        out.append('null')
    elif v is True:
        out.append('true')
    elif v is False:
        out.append('false')
    elif isinstance(v, int):
        out.append(str(v))
    elif isinstance(v, str):
        out.append('"' + escape(v) + '"')
    elif isinstance(v, list):
        out.append('[')
        for i, x in enumerate(v):
            if i:
                out.append(',')
            dump(x, out)
        out.append(']')
    else:
        out.append('{')
        first = True
        for k in sorted(v):
            if not first:
                out.append(',')
            first = False
            out.append('"' + escape(k) + '":')
            dump(v[k], out)
        out.append('}')


def main():
    raw = sys.stdin.buffer.read()
    try:
        text = raw.decode('utf-8')
        val = Parser(text).parse()
        out = []
        dump(val, out)
        result = ''.join(out)
    except (ParseError, UnicodeDecodeError, ValueError):
        result = 'ERR'
    sys.stdout.buffer.write(result.encode('utf-8') + b'\n')


if __name__ == '__main__':
    main()

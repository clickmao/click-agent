import sys
import json
import math


def solve(data):
    """对每个元素求平方根；负数输出 ERR，布尔按 0.0/1.0 处理。"""
    out = []
    for x in data:
        if isinstance(x, bool):
            x = float(x)
        x = float(x)
        if x < 0:
            out.append("ERR")
        else:
            out.append(repr(math.sqrt(x)))
    return out


def selftest():
    cases = [
        ([4, 9], ["2.0", "3.0"]),
        ([2], [repr(math.sqrt(2))]),
        ([0], ["0.0"]),
        ([-1, -4], ["ERR", "ERR"]),
        ([True, False], ["1.0", "0.0"]),
        ([], []),
    ]
    failed = 0
    for i, (inp, exp) in enumerate(cases, 1):
        got = solve(inp)
        ok = got == exp
        if not ok:
            failed += 1
        print("case%d %s" % (i, "PASS" if ok else "FAIL"))
    print("SELFTEST %s" % ("PASS" if failed == 0 else "FAIL"))
    return 0 if failed == 0 else 1


def main(argv):

        return selftest()

    if not raw.strip():
        data = []
    else:
        data = json.loads(raw)

    sys.stdout.write("\n".join(solve(data)) + "\n")
    return 0

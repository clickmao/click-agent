# -*- coding: utf-8 -*-
"""
JSON-lite 规范化器 (单文件, 仅标准库)

运行:
    python3 solution.py            # 从 stdin 读, 结果写 stdout
    python3 solution.py --selftest # 自检: 打印 PASS/FAIL, 退出码 0/非0

规则:
  - 合法值: null / true / false / 整数(可带一个负号, 无前导零, -0 归一为 0)
            / 字符串 / 数组 / 对象(键必须是字符串)
  - 字符串转义只允许 6 种: \\" \\\\ \\/ \\n \\t \\uXXXX
    \\uXXXX 解码后的码点必须 >= 0x20; 高位代理(D800-DBFF)必须紧跟低位代理
    (DC00-DFFF)并合并为一个字符; 孤立代理、裸控制字符(码点 < 0x20)一律非法
  - 非法输入(空输入 / 语法错 / 多余尾随内容 / 非法转义 / 非法数字)只输出 ERR
  - 合法输入输出其规范化形式: 去掉所有非必要空白, 字符串最小转义
"""

import sys

WS = ' \t\n\r'
HEX = '0123456789abcdefABCDEF'


class ParseError(Exception):
    pass


class JObj(object):
    __slots__ = ('pairs',)

    def __init__(self, pairs):
        self.pairs = pairs


class Parser(object):
    def __init__(self, s):
        self.s = s
        self.i = 0
        self.n = len(s)

    def err(self):
        raise ParseError()

    def skip_ws(self):
        s, n = self.s, self.n
        i = self.i
        while i < n and s[i] in WS:
            i += 1
        self.i = i

    def parse(self):
        v = self.value()
        self.skip_ws()
        if self.i != self.n:
            self.err()
        return v

    def value(self):
        self.skip_ws()
        if self.i >= self.n:
            self.err()
        c = self.s[self.i]
        if c == 'n':
            self.lit('null')
            return None
        if c == 't':
            self.lit('true')
            return True
        if c == 'f':
            self.lit('false')
            return False
        if c == '"':
            return self.string()
        if c == '[':
            return self.array()
        if c == '{':
            return self.obj()
        if c == '-' or ('0' <= c <= '9'):
            return self.number()
        self.err()

    def lit(self, w):
        if self.s.startswith(w, self.i):
            self.i += len(w)
        else:
            self.err()

    def number(self):
        s, n = self.s, self.n
        start = self.i
        if s[self.i] == '-':
            self.i += 1
            if self.i >= n or not ('0' <= s[self.i] <= '9'):
                self.err()
        if s[self.i] == '0':
            self.i += 1
            if self.i < n and '0' <= s[self.i] <= '9':
                self.err()
        else:
            while self.i < n and '0' <= s[self.i] <= '9':
                self.i += 1
        return int(s[start:self.i])

    def hex4(self, idx):
        if idx + 4 > self.n:
            self.err()
        chunk = self.s[idx:idx + 4]
        for ch in chunk:
            if ch not in HEX:
                self.err()
        return int(chunk, 16)

    def string(self):
        s, n = self.s, self.n
        self.i += 1  # 跳过开引号
        out = []
        while True:
            if self.i >= n:
                self.err()
            c = s[self.i]
            if c == '"':
                self.i += 1
                return ''.join(out)
            if c == '\\':
                self.i += 1
                if self.i >= n:
                    self.err()
                e = s[self.i]
                if e == '"':
                    out.append('"')
                    self.i += 1
                elif e == '\\':
                    out.append('\\')
                    self.i += 1
                elif e == '/':
                    out.append('/')
                    self.i += 1
                elif e == 'n':
                    out.append('\n')
                    self.i += 1
                elif e == 't':
                    out.append('\t')
                    self.i += 1
                elif e == 'u':
                    cp = self.hex4(self.i + 1)
                    self.i += 5
                    if 0xD800 <= cp <= 0xDBFF:
                        if s.startswith('\\u', self.i):
                            lo = self.hex4(self.i + 2)
                            if 0xDC00 <= lo <= 0xDFFF:
                                self.i += 6
                                out.append(chr(0x10000 + ((cp - 0xD800) << 10) +
                                               (lo - 0xDC00)))
                            else:
                                self.err()
                        else:
                            self.err()
                    elif 0xDC00 <= cp <= 0xDFFF:
                        self.err()
                    else:
                        if cp < 0x20:
                            self.err()
                        out.append(chr(cp))
                else:
                    self.err()
            else:
                if ord(c) < 0x20:
                    self.err()
                out.append(c)
                self.i += 1

    def array(self):
        self.i += 1  # '['
        res = []
        self.skip_ws()
        if self.i < self.n and self.s[self.i] == ']':
            self.i += 1
            return res
        while True:
            res.append(self.value())
            self.skip_ws()
            if self.i >= self.n:
                self.err()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
                continue
            if c == ']':
                self.i += 1
                return res
            self.err()

    def obj(self):
        self.i += 1  # '{'
        pairs = []
        self.skip_ws()
        if self.i < self.n and self.s[self.i] == '}':
            self.i += 1
            return JObj(pairs)
        while True:
            self.skip_ws()
            if self.i >= self.n or self.s[self.i] != '"':
                self.err()
            k = self.string()
            self.skip_ws()
            if self.i >= self.n or self.s[self.i] != ':':
                self.err()
            self.i += 1
            v = self.value()
            pairs.append((k, v))
            self.skip_ws()
            if self.i >= self.n:
                self.err()
            c = self.s[self.i]
            if c == ',':
                self.i += 1
                continue
            if c == '}':
                self.i += 1
                return JObj(pairs)
            self.err()


def esc(s):
    out = []
    for ch in s:
        if ch == '"':
            out.append('\\"')
        elif ch == '\\':
            out.append('\\\\')
        elif ch == '\n':
            out.append('\\n')
        elif ch == '\t':
            out.append('\\t')
        elif ord(ch) < 0x20:
            out.append('\\u%04x' % ord(ch))
        else:
            out.append(ch)
    return ''.join(out)


def dump(v):
    if v is None:
        return 'null'
    if v is True:
        return 'true'
    if v is False:
        return 'false'
    if isinstance(v, int):
        return str(v)
    if isinstance(v, str):
        return '"' + esc(v) + '"'
    if isinstance(v, list):
        return '[' + ','.join(dump(x) for x in v) + ']'
    if isinstance(v, JObj):
        return '{' + ','.join('"' + esc(k) + '":' + dump(val)
                              for k, val in v.pairs) + '}'
    raise ParseError()


def solve(text):
    try:
        return dump(Parser(text).parse())
    except (ParseError, RecursionError):
        return 'ERR'


def selftest():
    NL = '\n'
    TB = '\t'
    cases = [
        ('null', 'null'),
        ('  -0  ', '0'),
        ('[1, 2,3]', '[1,2,3]'),
        ('{"a": [true, null]}', '{"a":[true,null]}'),
        ('["\\u0002x"]', 'ERR'),
        ('"\\u0002"', 'ERR'),
        ('"\\u001f"', 'ERR'),
        ('"\\u0020"', '" "'),
        ('"\\u0041"', '"A"'),
        ('"a\\/b"', '"a/b"'),
        ('"a\\nb\\tc"', '"a\\nb\\tc"'),
        ('"\\ud83d\\ude00"', '"\U0001f600"'),
        ('"\\ud83d"', 'ERR'),
        ('"\\ude00"', 'ERR'),
        ('"\\ud83d\\u0041"', 'ERR'),
        ('"\\q"', 'ERR'),
        ('"\\u12"', 'ERR'),
        ('"\\u12g4"', 'ERR'),
        ('"' + NL + '"', 'ERR'),
        ('"a"' + TB, '"a"'),
        ('01', 'ERR'),
        ('-', 'ERR'),
        ('-01', 'ERR'),
        ('1.0', 'ERR'),
        ('[1,]', 'ERR'),
        ('{"a":1,}', 'ERR'),
        ('{a:1}', 'ERR'),
        ('{"a"1}', 'ERR'),
        ('', 'ERR'),
        ('   ', 'ERR'),
        ('true false', 'ERR'),
        ('[1] x', 'ERR'),
        ('[]', '[]'),
        ('{}', '{}'),
    ]
    ok = True
    for inp, exp in cases:
        got = solve(inp)
        if got != exp:
            ok = False
            sys.stdout.write('FAIL: %r -> %r expected %r\n' % (inp, got, exp))
    sys.stdout.write('PASS\n' if ok else 'FAIL\n')
    return 0 if ok else 1


def main(argv):
    if argv[1:2] == ['--selftest']:
        return selftest()
    sys.setrecursionlimit(20000)
    text = sys.stdin.read()
    sys.stdout.write(solve(text) + '\n')
    return 0


if __name__ == '__main__':
